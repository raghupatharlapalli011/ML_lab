    "# ---------------------------------------------------------\n",
    "# 3. Principal Component Analysis (PCA)\n",
    "# ---------------------------------------------------------\n",
    "# Features matrix combining Maths and Science\n",
    "X_pca = np.column_stack((maths, science))\n",
    "\n",
    "# Mean centering the data\n",
    "X_centered = X_pca - np.mean(X_pca, axis=0)\n",
    "\n",
    "# Performing PCA\n",
    "pca = PCA(n_components=2)\n",
    "pc_scores = pca.fit_transform(X_pca)\n",
    "\n",
    "print(\"=== 2. Principal Component Analysis (PCA) ===\")\n",
    "print(\"Eigenvalues (Variance Explained per PC):\")\n",
